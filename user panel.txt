  <script>
    lucide.createIcons();

    const adModal = document.getElementById('ad-modal');
    const adCountdown = document.getElementById('ad-countdown');
    const adStatusTitle = document.getElementById('ad-status-title');
    const adStatusDesc = document.getElementById('ad-status-desc');
    
    const playerModal = document.getElementById('player-modal');
    const videoPlayer = document.getElementById('main-video-player');
    const videoSource = document.getElementById('video-source');
    const playingVideoTitle = document.getElementById('playing-video-title');

    // Ads logic before playing video
    async function playVideoWithAds(title, videoUrl) {
      adModal.classList.remove('hidden');
      
      // Step 1: Trigger 1st Ad
      adStatusTitle.textContent = "১ম এড প্লে হচ্ছে...";
      adStatusDesc.textContent = "অনুগ্রহ করে এডটি দেখুন...";
      showSdkAd();
      await startCountdown(5);

      // Step 2: Trigger 2nd Ad
      adStatusTitle.textContent = "২য় এড প্লে হচ্ছে...";
      adStatusDesc.textContent = "ভিডিওটি লোড হচ্ছে, শেষ এডটি সম্পন্ন করুন...";
      showSdkAd();
      await startCountdown(5);

      // Step 3: Open Video Player
      adModal.classList.add('hidden');
      openVideoPlayer(title, videoUrl);
    }

    // Call Updated SDK Ad function safely
    function showSdkAd() {
      if (typeof window.show_12004088 === 'function') {
        try {
          window.show_12004088();
        } catch (e) {
          console.log("Ad Load Error");
        }
      }
    }

    // Countdown Timer Helper
    function startCountdown(seconds) {
      return new Promise((resolve) => {
        let timeLeft = seconds;
        adCountdown.textContent = timeLeft;
        const timer = setInterval(() => {
          timeLeft--;
          adCountdown.textContent = timeLeft;
          if (timeLeft <= 0) {
            clearInterval(timer);
            resolve();
          }
        }, 1000);
      });
    }

    // Open Main Video Player
    function openVideoPlayer(title, videoUrl) {
      playingVideoTitle.textContent = title;
      videoSource.src = videoUrl;
      videoPlayer.load();
      playerModal.classList.remove('hidden');
      videoPlayer.play();
    }

    // Close Player & Stop Video
    function closePlayer() {
      videoPlayer.pause();
      videoPlayer.currentTime = 0;
      playerModal.classList.add('hidden');
    }
  </script>
